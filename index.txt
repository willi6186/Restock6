    <script>
        const { createApp, ref, computed } = Vue;

        // 登入密碼設定
        const AUTH_CONFIG = {
            username: "admin",
            password: "wms1234"
        };

        createApp({
            setup() {
                // Portal & Auth State
                const isLoggedIn = ref(false);
                const currentUser = ref("");
                const currentTab = ref("home");
                const loginForm = ref({ username: "", password: "" });
                const loginError = ref("");

                // App State
                const stockData = ref([]);
                const salesMap = ref({});
                const putawayMap = ref({});
                const putawayCount = ref(0);
                const excludeInput = ref("TP01, Room, RM, BE, BD, BF, BG, CA, CP");
                const thresholdPercent = ref(10);
                const search = ref("");
                const sortBySales = ref(false);

                // 登入/登出處理
                const handleLogin = () => {
                    loginError.value = "";
                    if (loginForm.value.username === AUTH_CONFIG.username && loginForm.value.password === AUTH_CONFIG.password) {
                        isLoggedIn.value = true;
                        currentUser.value = loginForm.value.username;
                        loginForm.value = { username: "", password: "" };
                    } else {
                        loginError.value = "帳號或密碼錯誤 (預設: admin / wms1234)";
                    }
                };

                const handleLogout = () => {
                    isLoggedIn.value = false;
                    currentUser.value = "";
                    currentTab.value = "home";
                };

                const excludeList = computed(() => {
                    return excludeInput.value.split(',').map(s => s.trim().toLowerCase()).filter(s => s.length > 0);
                });

                const getLocTier = (loc) => {
                    const m = String(loc).trim().match(/[-_]?(\d+)$/);
                    return m ? parseInt(m[1], 10) : 99;
                };

                const parseFile = (file, callback) => {
                    const filename = file.name.toLowerCase();
                    if (filename.endsWith('.xlsx') || filename.endsWith('.xls')) {
                        const reader = new FileReader();
                        reader.onload = (evt) => {
                            const bstr = evt.target.result;
                            const wb = XLSX.read(bstr, { type: 'binary' });
                            const wsname = wb.SheetNames[0];
                            const ws = wb.Sheets[wsname];
                            callback(XLSX.utils.sheet_to_json(ws, { defval: "" }));
                        };
                        reader.readAsBinaryString(file);
                    } else {
                        Papa.parse(file, {
                            header: true,
                            skipEmptyLines: true,
                            complete: (results) => { callback(results.data); }
                        });
                    }
                };

                const handleStockUpload = (e) => {
                    const file = e.target.files[0];
                    if (!file) return;
                    parseFile(file, (data) => { stockData.value = data; });
                };

                const handleSalesUpload = (e) => {
                    const file = e.target.files[0];
                    if (!file) return;
                    parseFile(file, (data) => {
                        const map = {};
                        data.forEach(row => {
                            const key = String(row['Barcode'] || row['條碼'] || row['SKU编码'] || row['SKU'] || '').trim();
                            const qty = parseFloat(row['出庫數量'] || row['出货数量'] || row['出货量'] || row['数量'] || row['Qty'] || 1) || 0;
                            if (key) map[key] = (map[key] || 0) + qty;
                        });
                        salesMap.value = map;
                    });
                };

                const handlePutawayUpload = (e) => {
                    const file = e.target.files[0];
                    if (!file) return;
                    parseFile(file, (data) => {
                        const map = {};
                        let count = 0;

                        data.forEach(row => {
                            const loc = String(row['库位'] || row['庫位'] || row['Location'] || '').trim();
                            const barcode = String(row['Barcode'] || row['條碼'] || '').trim();
                            const sku = String(row['SKU编码'] || row['SKU編碼'] || row['SKU'] || '').trim();
                            const opType = String(row['入庫上架'] || row['入库上架'] || row['作業類型'] || row['Type'] || '').trim();
                            const rawTime = row['操作時間'] || row['操作时间'] || row['PutawayTime'] || row['Date'] || row['時間'];

                            const isPutawayAction = !opType || /上架|入庫|入库|putaway|inbound/i.test(opType);

                            if (loc && (barcode || sku) && rawTime && isPutawayAction) {
                                const parsedDate = new Date(rawTime);
                                const timeStamp = isNaN(parsedDate.getTime()) ? 0 : parsedDate.getTime();

                                const itemKeys = [];
                                if (barcode) itemKeys.push(`${loc}_${barcode}`);
                                if (sku) itemKeys.push(`${loc}_${sku}`);

                                itemKeys.forEach(mapKey => {
                                    if (!map[mapKey] || timeStamp < map[mapKey].timeStamp) {
                                        map[mapKey] = { timeStamp, rawTimeStr: String(rawTime) };
                                    }
                                });
                                count++;
                            }
                        });

                        putawayMap.value = map;
                        putawayCount.value = count;
                    });
                };

                const filteredStockData = computed(() => {
                    if (!excludeList.value.length) return stockData.value;
                    return stockData.value.filter(row => {
                        const loc = String(row['库位'] || row['庫位'] || '').toLowerCase();
                        return !excludeList.value.some(kw => loc.includes(kw));
                    });
                });

                const skuSummary = computed(() => {
                    const map = {};
                    filteredStockData.value.forEach(row => {
                        const barcode = String(row['Barcode'] || row['條碼'] || '').trim();
                        const sku = String(row['SKU编码'] || row['SKU編碼'] || row['SKU'] || '').trim();
                        const loc = String(row['库位'] || row['庫位'] || '').trim();
                        const availStock = parseFloat(row['可用库存'] || row['可用庫存'] || 0) || 0;
                        const pendingStock = parseFloat(row['已分配待出库存'] || row['已分配待出庫存'] || 0) || 0;

                        if (!barcode) return;

                        if (!map[barcode]) {
                            map[barcode] = { 
                                barcode, sku, 
                                name: row['中文名称'] || row['中文名稱'] || row['品名'] || '', 
                                totalStock: 0, groundStock: 0, highStock: 0,
                                allocatedPending: 0, highLocations: []
                            };
                        }

                        map[barcode].totalStock += availStock;
                        map[barcode].allocatedPending += pendingStock;

                        const isGround = loc.endsWith('-01') || loc.endsWith('01');

                        if (isGround) {
                            map[barcode].groundStock += availStock;
                        } else {
                            map[barcode].highStock += availStock;
                            if (availStock > 0) {
                                const pKeyBarcode = `${loc}_${barcode}`;
                                const pKeySKU = `${loc}_${sku}`;
                                const pInfo = putawayMap.value[pKeyBarcode] || putawayMap.value[pKeySKU];

                                map[barcode].highLocations.push({
                                    loc: loc,
                                    qty: availStock,
                                    tier: getLocTier(loc),
                                    opTimeStamp: pInfo ? pInfo.timeStamp : Infinity,
                                    opTimeStr: pInfo ? pInfo.rawTimeStr : null
                                });
                            }
                        }
                    });

                    const targetRatio = thresholdPercent.value / 100;

                    return Object.values(map).map(item => {
                        const groundRatio = item.totalStock > 0 ? (item.groundStock / item.totalStock) : 0;
                        const needReplenish = (groundRatio < targetRatio) && (item.highStock > 0);
                        const yesterdaySales = salesMap.value[item.barcode] || salesMap.value[item.sku] || 0;

                        let diffQty = 0;
                        let pickPlan = [];

                        if (needReplenish) {
                            const ratioDiff = targetRatio - groundRatio;
                            diffQty = Math.ceil(item.totalStock * ratioDiff);
                            diffQty = Math.min(diffQty, item.highStock);

                            const sortedHighLocs = item.highLocations.sort((a, b) => {
                                if (a.opTimeStamp !== b.opTimeStamp) return a.opTimeStamp - b.opTimeStamp;
                                return a.tier - b.tier;
                            });

                            let remainToPick = diffQty;
                            for (const locItem of sortedHighLocs) {
                                if (remainToPick <= 0) break;
                                const takeQty = Math.min(remainToPick, locItem.qty);
                                pickPlan.push({
                                    loc: locItem.loc,
                                    takeQty: takeQty,
                                    availQty: locItem.qty,
                                    opTime: locItem.opTimeStr
                                });
                                remainToPick -= takeQty;
                            }
                        }

                        return { 
                            ...item, groundRatio, needReplenish, yesterdaySales,
                            suggestedReplenish: diffQty, pickPlan
                        };
                    });
                });

                const totalSKUs = computed(() => Object.keys(skuSummary.value).length);
                const totalGroundStock = computed(() => skuSummary.value.reduce((acc, cur) => acc + cur.groundStock, 0));
                const replenishList = computed(() => skuSummary.value.filter(item => item.needReplenish));
                const totalSuggestedReplenish = computed(() => replenishList.value.reduce((acc, cur) => acc + cur.suggestedReplenish, 0));

                const filteredList = computed(() => {
                    let list = replenishList.value;
                    if (search.value.trim()) {
                        const q = search.value.trim().toLowerCase();
                        list = list.filter(i => i.barcode.toLowerCase().includes(q) || i.sku.toLowerCase().includes(q) || i.name.toLowerCase().includes(q));
                    }
                    if (sortBySales.value) {
                        return [...list].sort((a, b) => b.yesterdaySales - a.yesterdaySales);
                    }
                    return list;
                });

                const exportCSV = () => {
                    if (!replenishList.value.length) return;
                    const exportData = [];
                    replenishList.value.forEach(item => {
                        const locs = item.pickPlan.map(p => p.loc).join(', ');
                        exportData.push({
                            'Barcode': item.barcode,
                            'SKU編碼': item.sku,
                            '品名': item.name,
                            '昨日出貨量': item.yesterdaySales,
                            '總庫存': item.totalStock,
                            '地下可用庫存': item.groundStock,
                            '地下佔比': (item.groundRatio * 100).toFixed(1) + '%',
                            '建議下架總量': item.suggestedReplenish,
                            '建議下架庫位(FIFO)': locs
                        });
                    });

                    const csv = Papa.unparse(exportData);
                    const blob = new Blob(["\uFEFF" + csv], { type: 'text/csv;charset=utf-8;' });
                    const link = document.createElement("a");
                    link.href = URL.createObjectURL(blob);
                    link.setAttribute("download", `WMS_Replenishment_Task_FIFO_${new Date().toISOString().slice(0,10)}.csv`);
                    document.body.appendChild(link);
                    link.click();
                    document.body.removeChild(link);
                };

                return {
                    isLoggedIn, currentUser, currentTab, loginForm, loginError, handleLogin, handleLogout,
                    stockData, salesMap, putawayMap, putawayCount, excludeInput, thresholdPercent, search, sortBySales,
                    totalSKUs, totalGroundStock, replenishList, totalSuggestedReplenish, filteredList,
                    handleStockUpload, handleSalesUpload, handlePutawayUpload, exportCSV
                };
            }
        }).mount('#app');
    </script>
